# Core runtime dependencies for the Yoruba pronunciation assessment engine.

# Montreal Forced Aligner is NOT listed here: it ships Kaldi binaries and must be
# installed via conda. See INSTALL.md.

#   conda install -c conda-forge montreal-forced-aligner

numpy>=1.24
soundfile>=0.12
praat-parselmouth>=0.4.3

# The tone and vowel classifiers are scikit-learn estimators persisted with
# joblib. Both are imported lazily, so the package imports without them and then
# fails at first use -- which is worse than a missing dependency, so they are
# declared.
scikit-learn>=1.4
joblib>=1.3

# --- Vowel identity ----------------------------------------------------------
# Vowel identity uses a mean-pooled wav2vec2 layer (87.3% accuracy, macro F1
# 0.811) in place of the 39 hand-designed measurements it replaced (59.5%,
# 0.432). Optional: without torch, transformers, or the checkpoint, the engine
# logs a warning and disables vowel-identity checking. Omission detection, tone
# comparison and GOP all still work.
#
# The CPU build is enough -- 4.7x realtime, against forced alignment's ~35 s:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch>=2.0
transformers>=4.40
